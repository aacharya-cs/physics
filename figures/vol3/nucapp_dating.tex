% nucapp_dating: parent decays, daughter grows, their ratio dates the sample (Vol III Ch 5, fig:nucapp-dating)
% Closed system, no daughter at t = 0, stable daughter.  With s = t/T_{1/2}:
%   parent    N_X / N_{X,0} = 2^{-s}
%   daughter  N_Y / N_{X,0} = 1 - 2^{-s}
%   ratio     N_Y / N_X     = 2^{s} - 1   (= e^{lambda t} - 1), right axis
% The ratio is 1, 3, 7, 15 after 1, 2, 3, 4 half-lives.  (Source data: figs.py, fig 10.)
\begin{tikzpicture}
\begin{axis}[bookplot,
    width=10.4cm, height=5.6cm,
    xmin=0, xmax=4, ymin=0, ymax=1.25, domain=0:4,
    axis lines*=left,
    xlabel={time in half-lives, $t/\Thalf$},
    ylabel={fraction of $N_{X,0}$},
    xtick={1,2,3,4}, ytick={0.5,1},
    ylabel style={at={(axis description cs:0,1)},anchor=south,rotate=-90,yshift=2pt}]
  \addplot[cPrim] {2^(-x)};
  \addplot[cTeal] {1-2^(-x)};
  \node[cPrim,anchor=south west] at (axis cs:0.33,0.80) {parent, $N_X/N_{X,0}$};
  \node[cTeal,anchor=south west] at (axis cs:2.1,0.97) {daughter, $N_Y/N_{X,0}$};
  \draw[guide] (axis cs:1,0) -- (axis cs:1,0.5);
\end{axis}
\begin{axis}[bookplot,
    width=10.4cm, height=5.6cm,
    xmin=0, xmax=4, ymin=0, ymax=18.75, domain=0:4,
    axis x line=none, axis y line*=right,
    y axis line style={draw=cAcc},
    ytick={1,3,7,15}, yticklabel style={text=cAcc},
    every tick/.style={line width=0.5pt,draw=cAcc},
    ylabel={$N_Y/N_X$}, ylabel style={text=cAcc,at={(axis description cs:1,1)},anchor=south,rotate=-90,yshift=2pt}]
  \addplot[cAcc,dash pattern=on 4pt off 2pt] {2^x-1};
  \foreach \n in {1,2,3} {
    \edef\temp{\noexpand\node[dot,cAcc] at (axis cs:\n,{2^\n-1}) {};}\temp }
  \node[dot,cAcc] at (axis cs:4,15) {};
  \node[cAcc,anchor=east] at (axis cs:3.12,8.0) {ratio $N_Y/N_X=2^{t/\Thalf}-1$};
\end{axis}
\end{tikzpicture}
