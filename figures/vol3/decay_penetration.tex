% decay_penetration: penetrating power of alpha, beta and gamma radiation
% (Vol III Ch 2, fig:decay-penetration)
% alpha (cPlum): stopped by a sheet of paper (a few cm of air).
% beta (cPrim, electrons): stopped by a few mm of aluminium.
% gamma (cAcc, wavy): attenuated exponentially by lead; the intensity falls but
% never reaches exactly zero, so a weakened beam continues (drawn fainter).
% Colours: alpha cPlum, electron cPrim, gamma the shared wavy `photon' style.
\begin{tikzpicture}[
  ray/.style={vector},
  gamma/.style={vector,line cap=butt,photon}]
\draw[outline,fill=cRock!60] (-0.6,-1.4) rectangle (0.45,1.4);
\node[small label,align=center] at (-0.075,0) {source};
\fill[cAmber!30] (2.3,-1.6) rectangle (2.42,1.6); \draw[thin line] (2.3,-1.6) rectangle (2.42,1.6);
\node[small label,align=center,anchor=south] at (2.36,1.65) {paper};
\fill[cMoon] (4.4,-1.6) rectangle (4.75,1.6); \draw[thin line] (4.4,-1.6) rectangle (4.75,1.6);
\node[small label,align=center,anchor=south] at (4.575,1.65) {aluminium\\(a few mm)};
\fill[cRock] (7.0,-1.6) rectangle (8.0,1.6); \draw[thin line] (7.0,-1.6) rectangle (8.0,1.6);
\node[small label,align=center,anchor=south] at (7.5,1.65) {lead\\(several cm)};
\draw[ray,cPlum] (0.55,0.9) -- (2.28,0.9) node[pos=0.5,above,cPlum] {$\alpha$};
\draw[ray,cPrim] (0.55,0) -- (4.38,0) node[pos=0.25,above,cPrim] {$\beta$};
\draw[gamma] (0.55,-0.9) -- (6.98,-0.9);
\node[cAcc] at (1.38,-0.6) {$\gamma$};
\draw[gamma,opacity=0.45,thin line] (8.0,-0.9) -- (9.6,-0.9);
\node[small label,align=center,text=cAcc] at (9.0,-0.3) {weakened,\\not stopped};
\end{tikzpicture}
