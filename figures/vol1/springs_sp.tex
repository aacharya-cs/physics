% springs_sp: springs in series and in parallel (Ch 8, fig:springs)
% Each spring has natural length 1.0 cm (8 turns).  Hanging a load that pulls with F:
%   series:   k1 = 1.56 k2, so x1 = 0.45 cm, x2 = 0.70 cm (same F in both,
%             extensions add: x = x1 + x2)
%   parallel: k1 = 1.5 k2; both stretch by the same x = 0.5 cm, so they carry
%             F1 = 0.6F and F2 = 0.4F.  For the bar to stay level the load hangs
%             where F1 d1 = F2 d2: 0.4 cm from k1, 0.6 cm from k2 (springs 1 cm apart).
% Dashed ticks mark where each spring's lower end would be at natural length.
\begin{tikzpicture}[x=1cm,y=1cm]
  % a vertical coil spring from (#1,#2) down to (#1,#3): leads of 0.15 cm, 8 turns
  \def\vspring(#1,#2,#3){%
    \pgfmathsetmacro\p{((#2)-(#3)-0.3)/8}%
    \draw[cInk,thin line] (#1,#2) -- ++(0,-0.15)
      plot[domain=0:16*pi,samples=320,smooth,variable=\t]
        ({#1+0.16*sin(\t r)},{#2-0.15-\p*\t/(2*pi)-0.06*(1-cos(\t r))})
      -- (#1,#3);}
  % ------------------------------------------------------------- series ---
  \path[ground,draw=none] (-0.8,0) rectangle (0.8,0.16);
  \draw[outline] (-0.8,0) -- (0.8,0);
  \vspring(0,0,-1.45)
  \vspring(0,-1.45,-3.15)
  \node[point mass] at (0,-1.45) {};
  \node[anchor=west] at (0.25,-0.70) {$k_1$};
  \node[anchor=west] at (0.25,-2.25) {$k_2$};
  \draw[body] (-0.38,-3.67) rectangle (0.38,-3.15);
  \draw[force] (0,-3.67) -- ++(0,-0.8) node[anchor=west,pos=0.6] {$F$};
  % extensions, measured from the natural-length positions
  \draw[dimtick] (-0.2,-1.0) -- (-0.75,-1.0);
  \draw[dimtick] (-0.2,-1.45) -- (-0.75,-1.45);
  \draw[dim] (-0.62,-1.0) -- node[cSteel,anchor=east] {$x_1$} (-0.62,-1.45);
  \draw[dimtick] (-0.2,-2.45) -- (-0.75,-2.45);
  \draw[dimtick] (-0.4,-3.15) -- (-0.75,-3.15);
  \draw[dim] (-0.62,-2.45) -- node[cSteel,anchor=east] {$x_2$} (-0.62,-3.15);
  \node[anchor=north,font=\footnotesize,text=cSteel,align=center] at (0,-4.65)
    {series: same force,\\ extensions add};
  % ----------------------------------------------------------- parallel ---
  \begin{scope}[xshift=4.0cm]
    \path[ground,draw=none] (-1.0,0) rectangle (1.0,0.16);
    \draw[outline] (-1.0,0) -- (1.0,0);
    \vspring(-0.5,0,-1.5)
    \vspring(0.5,0,-1.5)
    \node[anchor=east] at (-0.75,-0.7) {$k_1$};
    \node[anchor=west] at (0.75,-0.7) {$k_2$};
    % rigid bar, and the load hung from its middle
    \draw[cInk,thin line,fill=cRock] (-0.8,-1.5) rectangle (0.8,-1.62);
    \draw[cInk,thin line] (-0.1,-1.62) -- (-0.1,-2.1);
    \draw[body] (-0.48,-2.62) rectangle (0.28,-2.1);
    \draw[force] (-0.1,-2.62) -- ++(0,-0.8) node[anchor=west,pos=0.6] {$F$};
    % the common extension
    \draw[dimtick] (0.7,-1.0) -- (1.35,-1.0);
    \draw[dimtick] (0.85,-1.5) -- (1.35,-1.5);
    \draw[dim] (1.2,-1.0) -- node[cSteel,anchor=west] {$x$} (1.2,-1.5);
    \node[anchor=north,font=\footnotesize,text=cSteel,align=center] at (0,-4.65)
      {parallel: same extension,\\ forces add};
  \end{scope}
\end{tikzpicture}
