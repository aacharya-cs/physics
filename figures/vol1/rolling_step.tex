% rolling_step: a rolling sphere meeting a step (Ch 15, fig:rollstep, step challenge)
% Drawn to the challenge's numbers at 10 cm per metre: R = 0.25 m -> 2.5 cm,
% h = 0.05 m -> 0.5 cm.  At impact the sphere touches the floor below its centre
% and the step's edge E, which is a horizontal distance sqrt(R^2-(R-h)^2) = 1.5 cm
% ahead.  The CM moves horizontally at height R, a perpendicular distance R-h above
% E; the line from E to the centre makes beta with the vertical, cos beta = (R-h)/R.
\begin{tikzpicture}[x=1cm,y=1cm]
  \def\R{2.5}\def\h{0.5}
  \pgfmathsetmacro\ex{sqrt(\R*\R-(\R-\h)*(\R-\h))}
  \coordinate (C) at (0,\R);
  \coordinate (E) at (\ex,\h);
  % floor and step
  \path[ground strip] (-2.9,-0.3) rectangle (\ex,0);
  \path[ground strip] (\ex,-0.3) rectangle (4.3,\h);
  \draw[outline] (-2.9,0) -- (\ex,0) -- (E) -- (4.3,\h);
  % the sphere
  \draw[planet,fill=cBlock!45,draw=cSteel] (C) circle[radius=\R];
  \pic at (C) {cm};
  % CM velocity and spin
  \draw[velocity] (C) -- ++(1.3,0) node[anchor=south,pos=0.75] {$v_0$};
  \draw[rotation,cFor] ($(C)+(150:0.85)$) arc[start angle=150,end angle=40,radius=0.85]
    node[anchor=south west,inner sep=1pt] {$\omega_0$};
  % the edge, the line from it to the centre, and beta
  \node[contact point] at (E) {};
  \node[anchor=north west] at (E) {$E$};
  \draw[cInk,thin line] (E) -- (C);
  \draw[guide] (E) -- ++(0,2.45);
  \draw[angle arc] ($(E)+(0,0.85)$) arc[start angle=90,end angle={90+acos((\R-\h)/\R)},radius=0.85];
  \node at ($(E)+({90+acos((\R-\h)/\R)/2}:1.12)$) {$\beta$};
  % R - h: perpendicular distance from E up to the CM's line of motion
  \draw[guide] ($(C)+(1.45,0)$) -- ++(2.65,0);
  \draw[dim] ($(E)+(1.6,0)$) -- node[anchor=west,cSteel] {$R-h$} ++(0,\R-\h);
  % step height
  \draw[dim] ($(E)+(2.3,-\h)$) -- node[anchor=west,cSteel] {$h$} ++(0,\h);
  % radius
  \draw[dim] (C) -- node[anchor=east,cSteel,pos=0.55] {$R$} (0,0);
\end{tikzpicture}
