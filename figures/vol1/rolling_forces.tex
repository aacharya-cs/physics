% rolling_forces: forces on a body rolling without slipping down an incline (Ch 15, fig:rollforces)
% Drawn for a uniform disc, c = 1/2, on theta = 25 degrees, forces to scale:
%   Mg = 2.8 cm,  N = Mg cos(theta) = 2.54 cm,
%   f = cMa = c/(1+c) Mg sin(theta) = (1/3)(0.423)Mg = 0.39 cm   (static, UP the slope).
% Mg and N act through the centre and have no moment arm; f at the contact gives the
% clockwise torque fR, so the angular acceleration alpha is clockwise while the centre
% accelerates DOWN the slope at a = g sin(theta)/(1+c)  (a is not to the force scale).
\begin{tikzpicture}[x=1cm,y=1cm]
  \def\th{25}
  \def\Lx{9.2}
  \def\Rr{0.9}
  \def\W{2.8}                                   % length of Mg
  \def\c{0.5}
  \pgfmathsetmacro\H{\Lx*tan(\th)}
  \pgfmathsetmacro\Nl{\W*cos(\th)}
  \pgfmathsetmacro\fl{\c/(1+\c)*\W*sin(\th)}
  % ground and incline (high on the left, the body rolls down to the right)
  \fill[ground strip] (-0.3,0) rectangle (\Lx+0.5,-0.18);
  \draw[outline] (-0.3,0) -- (\Lx+0.5,0);
  \filldraw[fill=cSlope,outline] (0,0) -- (0,\H) -- (\Lx,0) -- cycle;
  % angle of the incline
  \draw[angle arc] (\Lx-1.3,0) arc[start angle=180,end angle=180-\th,radius=1.3];
  \node at ($(\Lx,0)+(180-\th/2:1.62)$) {$\theta$};
  % everything on the slope, in the slope's own frame: x' down the slope, y' outward
  \begin{scope}[shift={(0,\H)},rotate=-\th]
    \coordinate (C) at (3.7,0);                 % contact point
    \coordinate (O) at (3.7,\Rr);               % centre
    \draw[body,fill=cBlock!45] (O) circle[radius=\Rr];
    % angular acceleration, clockwise, on the side clear of the force lines
    \draw[-smallhead,thin line,cPrim] ($(O)+(210:0.55)$) arc[start angle=210,end angle=112,radius=0.55];
    \node[cPrim] at ($(O)+(160:0.3)$) {$\alpha$};
    % acceleration of the centre, down the slope
    \draw[accel] ($(O)+(\Rr+0.35,0)$) -- ++(1.2,0) node[midway,above=1pt,cPrim] {$a$};
    % normal force at the contact, through the centre
    \draw[force] (C) -- ++(0,\Nl) node[anchor=south west,inner sep=1pt] {$N$};
    % static friction at the contact, up the slope
    \draw[force] (C) -- ++(-\fl,0);
    \node[cCrim,anchor=north] at ($(C)+(-\fl-0.05,-0.08)$) {$f$};
  \end{scope}
  \node[point mass] at (O) {};
  % weight from the centre, straight down
  \draw[force] (O) -- ++(0,-\W) node[anchor=west,pos=0.93] {$Mg$};
\end{tikzpicture}
