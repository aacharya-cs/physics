% thermo_isoadia: doubling the volume isothermally and adiabatically (Vol II Ch 4, fig:thermo-isoadia)
% Two moles of a monatomic ideal gas (gamma = 5/3) start at (V1, P1), T1 = 300 K.
% Units: V in V1, P in P1.  Isotherm P = 1/V ; adiabat P = V^(-5/3).  Both end at V = 2.
%   isothermal work  W = nRT ln 2           = 3458 J   (area under the upper curve)
%   adiabatic work   W = nR(T1-T2)/(gamma-1) = 2769 J   (area under the lower curve)
% Ratio 2769/3458 = 0.801 = (1 - 2^(-2/3))/((2/3) ln 2), as the areas show.
% End pressures: isotherm 0.5 P1, adiabat 2^(-5/3) = 0.315 P1 (end temperature 189 K).
\begin{tikzpicture}
\begin{axis}[bookplot,
    width=7.6cm, height=4.6cm, scale only axis,
    xmin=0, xmax=2.5, ymin=0, ymax=1.3,
    xlabel={volume $V$}, ylabel={pressure $P$},
    ylabel style={at={(axis description cs:0,1)},anchor=south west,rotate=-90,xshift=-4pt},
    xtick={1,2}, xticklabels={$V_1$,$2V_1$}, ytick={1}, yticklabels={$P_1$}]
  \addplot[draw=none,fill=cTeal!16,domain=1:2] {x^(-5/3)} \closedcycle;
  \addplot[draw=none,fill=cPrim!16,domain=1:2] {1/x} \closedcycle;
  \addplot[draw=none,fill=cTeal!16,domain=1:2] {x^(-5/3)} \closedcycle;
  \draw[guide] (axis cs:0,1) -- (axis cs:1,1);
  \draw[guide] (axis cs:2,0) -- (axis cs:2,0.315);
  \addplot[cPrim,domain=0.85:2.15] {1/x};
  \addplot[cTeal,domain=0.9:2.15] {x^(-5/3)};
  \node[dot] at (axis cs:1,1) {};
  \node[dot] at (axis cs:2,0.5) {};
  \node[dot] at (axis cs:2,0.315) {};
  \node[cPrim,anchor=west] at (axis cs:2.2,0.465) {isothermal, $\SI{300}{K}$};
  \node[cTeal,anchor=west] at (axis cs:2.2,0.28) {adiabatic, ends at $\SI{189}{K}$};
  \node[cTeal] at (axis cs:1.45,0.2) {$\SI{2769}{J}$};
  \draw[leader,cPrim] (axis cs:1.55,0.55) -- (axis cs:1.75,0.95)
     node[anchor=south west,inner sep=1pt] {extra $\SI{689}{J}$: total $\SI{3458}{J}$};
\end{axis}
\end{tikzpicture}
