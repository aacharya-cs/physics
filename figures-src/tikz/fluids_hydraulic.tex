% fluids_hydraulic: the hydraulic lift (Vol II Ch 2, fig:fluids-hydraulic)
% Two pistons, areas A1 (narrow, width 0.6) and A2 (wide, width 2.4) on one
% enclosed liquid; both piston faces at the same level, so the pressure under each
% is the same P = F1/A1 (Pascal's law).  The force on the large piston is
% F2 = P A2 = F1 (A2/A1).  Incompressible liquid: the volume pushed down on the left,
% A1 d1, reappears on the right, A2 d2, so d2 = d1 A1/A2 and F1 d1 = F2 d2.
% Widths 0.6 and 2.4 give an area ratio of 16 for circular pistons; arrows are not to scale.
\begin{tikzpicture}[x=1cm,y=1cm,
    water/.style={fill=cPrim!10,draw=none}]
  % liquid: left column, horizontal duct, right column
  \fill[water] (-0.3,-1.9) rectangle (0.3,0);          % left column under piston 1
  \fill[water] (-0.3,-1.9) rectangle (4.2,-1.3);       % duct
  \fill[water] (1.8,-1.9) rectangle (4.2,0);           % right column under piston 2
  % vessel walls
  \draw[outline] (-0.3,0.9) -- (-0.3,-1.9) -- (4.2,-1.9) -- (4.2,0.9);
  \draw[outline] (0.3,0.9) -- (0.3,-1.3) -- (1.8,-1.3) -- (1.8,0.9);
  % pistons
  \draw[body] (-0.3,0) rectangle (0.3,0.22);
  \draw[body] (1.8,0) rectangle (4.2,0.22);
  % load on the big piston
  \draw[fill=cMoon,draw=cInk,thin line] (2.35,0.22) rectangle (3.65,0.95);
  \node at (3,0.585) {load};
  % forces: the push on the small piston, and the liquid's push up on the large one
  \draw[force] (0,1.25) -- (0,0.24) node[pos=0.1,anchor=west,xshift=1pt] {$F_1$};
  \draw[force] (3,-0.95) -- (3,-0.02) node[pos=0.25,anchor=west] {$F_2=PA_2$};
  % areas
  \node[anchor=east,cSteel] at (-0.35,0.11) {$A_1$};
  \node[anchor=west,cSteel] at (4.25,0.11) {$A_2$};
  % the same pressure under both pistons (same level)
  \draw[guide] (-0.3,-0.45) -- (0.3,-0.45);
  \draw[guide] (1.8,-0.45) -- (4.2,-0.45);
  \node[cInk,anchor=west] at (-0.25,-1.6) {same $P=F_1/A_1$ under both pistons};
  % where the small piston goes: down a distance d1 (the large one rises far less)
  \draw[body,ghost,hidden] (-0.3,-0.7) rectangle (0.3,-0.48);
  \draw[dim] (-0.75,0) -- (-0.75,-0.7) node[midway,anchor=east,cSteel] {$d_1$};
\end{tikzpicture}
