% normal_force: the normal force takes whatever value the situation needs (Ch 7, fig:normal)
% Top row: three situations.  Bottom row: the block isolated, with every force drawn
% from its centre (the particle model of a free-body diagram), lengths to scale:
%   flat:     N = mg                                        (mg = 1.2 cm)
%   incline:  N = mg cos(theta), theta = 30 deg  -> 1.04 cm  (the slope is smooth;
%             the along-slope part of mg is unbalanced and accelerates the block)
%   lift accelerating up with a = 0.4 g:  N = m(g + a) = 1.4 mg -> 1.68 cm
% In the incline diagram the angle between mg and the inward normal is theta.
\begin{tikzpicture}[x=1cm,y=1cm]
  \def\W{1.2}
  \def\th{30}
  \def\xa{0}\def\xb{4.7}\def\xc{9.4}
  \def\yS{2.65}                                 % floor level of the scenes
  % ================= scenes
  % flat
  \path[ground,draw=none] (\xa-1.5,\yS-0.12) rectangle (\xa+1.5,\yS);
  \draw[outline] (\xa-1.5,\yS) -- (\xa+1.5,\yS);
  \draw[body] (\xa-0.55,\yS) rectangle (\xa+0.55,\yS+0.6);
  % incline
  \begin{scope}[shift={(\xb-1.5,\yS)}]
    \pgfmathsetmacro\Hh{3*tan(\th)}
    \fill[cSlope] (0,0) -- (3,0) -- (3,\Hh) -- cycle;
    \draw[outline] (0,0) -- (3,\Hh) -- (3,0) -- cycle;
    \draw[angle arc] (0.75,0) arc[start angle=0,end angle=\th,radius=0.75];
    \node at (\th/2:1.02) {$\theta$};
    \begin{scope}[shift={(\th:1.95)},rotate=\th]
      \draw[body] (-0.4,0) rectangle (0.4,0.5);
    \end{scope}
  \end{scope}
  % lift accelerating upward
  \draw[thin line,cInk] (\xc,\yS+1.6) -- (\xc,\yS+2.0);
  \draw[outline] (\xc-0.95,\yS) rectangle (\xc+0.95,\yS+1.6);
  \draw[body] (\xc-0.45,\yS) rectangle (\xc+0.45,\yS+0.6);
  \draw[accel] (\xc+1.3,\yS+0.4) -- ++(0,0.9) node[anchor=west] {$a$};
  % ================= free-body diagrams
  % flat
  \begin{scope}[shift={(\xa,0.3)}]
    \draw[body] (-0.5,-0.3) rectangle (0.5,0.3);
    \draw[force] (0,0) -- (0,\W) node[anchor=west] {$N$};
    \draw[force] (0,0) -- (0,-\W) node[anchor=west] {$mg$};
    \node at (0,-1.75) {$N=mg$};
  \end{scope}
  % incline
  \begin{scope}[shift={(\xb,0.3)}]
    \begin{scope}[rotate=\th]
      \draw[body] (-0.5,-0.3) rectangle (0.5,0.3);
    \end{scope}
    \pgfmathsetmacro\Nl{\W*cos(\th)}
    \draw[guide] (0,0) -- (-90+\th:1.25);
    \draw[angle arc] (-90:0.75) arc[start angle=-90,end angle=-90+\th,radius=0.75];
    \node at (-90+\th/2:0.98) {$\theta$};
    \draw[force] (0,0) -- (90+\th:\Nl) node[anchor=south east,inner sep=1pt] {$N$};
    \draw[force] (0,0) -- (0,-\W) node[anchor=east] {$mg$};
    \node at (0,-1.75) {$N=mg\cos\theta$};
  \end{scope}
  % lift
  \begin{scope}[shift={(\xc,0.3)}]
    \draw[body] (-0.5,-0.3) rectangle (0.5,0.3);
    \draw[force] (0,0) -- (0,1.4*\W) node[anchor=west] {$N$};
    \draw[force] (0,0) -- (0,-\W) node[anchor=west] {$mg$};
    \draw[accel] (0.95,-0.45) -- ++(0,0.9) node[anchor=west] {$a$};
    \node at (0,-1.75) {$N=m(g+a)$};
  \end{scope}
\end{tikzpicture}
