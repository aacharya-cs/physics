% thermal_water_density: anomalous expansion of water (Vol II Ch 3, fig:water-density)
% Near its maximum the density of liquid water at 1 atm is well fitted by a parabola
%   rho(t_C) = 999.97 - 0.0081 (t_C - 4)^2   kg/m^3,
% which reproduces the tabulated values (999.84 at 0 C, 999.97 at 4 C, 999.70 at 10 C)
% to within 0.02 kg/m^3.  Ice at 0 C (917 kg/m^3) lies far below the plotted range.
\begin{tikzpicture}
\begin{axis}[bookplot,
    width=9.0cm, height=5.0cm,
    xmin=0, xmax=11, ymin=999.60, ymax=1000.02,
    xtick={0,2,4,6,8,10},
    ytick={999.7,999.8,999.9},
    yticklabel style={/pgf/number format/.cd,fixed,precision=1,1000 sep={}},
    xlabel={$t_C$ (\si{\celsius})},
    ylabel={density $\rho$ (\si{kg/m^3})},
    ylabel style={at={(axis description cs:0,1)},anchor=south west,rotate=-90,xshift=-14pt}]
  \addplot[cPrim,domain=0:10.5] {999.97-0.0081*(x-4)^2};
  \draw[guide] (axis cs:4,999.60) -- (axis cs:4,999.97);
  \node[dot,cPrim] at (axis cs:4,999.97) {};
  \node[anchor=south,align=center] at (axis cs:4,999.975) {densest at \SI{4}{\celsius}};
  \node[anchor=north west,align=left,font=\footnotesize] at (axis cs:0.6,999.78)
    {heating\\ shrinks it};
  \node[anchor=north west,align=left,font=\footnotesize] at (axis cs:5.4,999.78)
    {heating\\ expands it\\ (normal)};
\end{axis}
\end{tikzpicture}
