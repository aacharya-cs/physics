% fluids_contactangle: the angle of contact (Vol II Ch 2, fig:fluids-contactangle)
% At the contact point C the liquid surface leaves the wall.  The angle of contact
% theta is the angle from the wall (pointing down, into the liquid) to the tangent
% of the liquid surface at C, swept through the liquid.
%  Left, wetting (theta = 30 deg): the tangent points down and away, direction -60 deg;
%  the surface curves up against the wall (concave meniscus) and levels off below C.
%  Right, non-wetting (theta = 140 deg): the tangent points up and away, direction
%  +50 deg; the surface is depressed at the wall and levels off above C.
% Each surface is a Bezier leaving C along its tangent and arriving level.
\begin{tikzpicture}[x=1cm,y=1cm,
    liquid/.style={fill=cPrim!12,draw=none},
    glass/.style={fill=cMoon,draw=cSteel,thin line}]
  % ---------------- wetting ----------------
  \begin{scope}
    \coordinate (C) at (0,2.0);
    \def\th{30}
    \fill[liquid] (C) .. controls ($(C)+({0.55*sin(\th)},{-0.55*cos(\th)})$) and (1.0,1.15)
       .. (2.6,1.15) -- (2.6,0) -- (0,0) -- cycle;
    \draw[cPrim,construction] (C) .. controls ($(C)+({0.55*sin(\th)},{-0.55*cos(\th)})$) and (1.0,1.15)
       .. (2.6,1.15);
    \draw[glass] (-0.35,0) rectangle (0,2.9);
    % tangent and the angle
    \draw[cSteel,thin line] ($(C)+({-0.25*sin(\th)},{0.25*cos(\th)})$) -- ($(C)+({1.35*sin(\th)},{-1.35*cos(\th)})$);
    \draw[angle arc] ($(C)+(0,-0.62)$) arc[start angle=-90,end angle={-90+\th},radius=0.62];
    \node at ($(C)+({0.88*sin(\th/2)},{-0.88*cos(\th/2)})$) {$\theta$};
    \node[contact point] at (C) {};
    \node[anchor=west,cSteel] at ($(C)+({1.25*sin(\th)},{-1.25*cos(\th)})+(0.05,0)$) {tangent};
    \node[cPrim] at (1.75,0.55) {liquid};
    \node[anchor=south,cSteel] at (-0.175,2.9) {glass};
    \node[align=center,anchor=north] at (1.15,-0.12) {wetting, $\theta<90^\circ$\\ (water on glass)};
  \end{scope}
  % ---------------- non-wetting ----------------
  \begin{scope}[xshift=4.6cm]
    \coordinate (C) at (0,1.15);
    \def\th{140}
    \fill[liquid] (C) .. controls ($(C)+({0.5*sin(\th)},{-0.5*cos(\th)})$) and (1.0,1.95)
       .. (2.6,1.95) -- (2.6,0) -- (0,0) -- cycle;
    \draw[cPrim,construction] (C) .. controls ($(C)+({0.5*sin(\th)},{-0.5*cos(\th)})$) and (1.0,1.95)
       .. (2.6,1.95);
    \draw[glass] (-0.35,0) rectangle (0,2.9);
    \draw[cSteel,thin line] ($(C)+({-0.25*sin(\th)},{0.25*cos(\th)})$) -- ($(C)+({1.3*sin(\th)},{-1.3*cos(\th)})$);
    \draw[angle arc] ($(C)+(0,-0.5)$) arc[start angle=-90,end angle={-90+\th},radius=0.5];
    \node at ($(C)+(0.7,-0.32)$) {$\theta$};
    \node[contact point] at (C) {};
    \node[anchor=south west,cSteel,inner sep=1pt] at ($(C)+({1.3*sin(\th)},{-1.3*cos(\th)})$) {tangent};
    \node[cPrim] at (1.75,0.55) {liquid};
    \node[anchor=south,cSteel] at (-0.175,2.9) {glass};
    \node[align=center,anchor=north] at (1.15,-0.12) {non-wetting, $\theta>90^\circ$\\ (mercury on glass)};
  \end{scope}
\end{tikzpicture}
