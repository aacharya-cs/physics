% thermo_twopaths: work depends on the path (Vol II Ch 4, fig:thermo-twopaths)
% States i = (V, P) = (1e-3 m^3, 2e5 Pa) and f = (2e-3 m^3, 1e5 Pa).
% Path 1: isobaric expansion at 2e5 Pa, then isochoric drop:  W1 = (2e5)(1e-3) = 200 J.
% Path 2: isochoric drop first, then isobaric expansion at 1e5 Pa:  W2 = (1e5)(1e-3) = 100 J.
% W is the area under each path (V axis to the path); the two areas are shaded,
% path 2's lying inside path 1's; the upper strip is the difference W1 - W2.  Axis units: V in 1e-3 m^3, P in 1e5 Pa.
\begin{tikzpicture}
\begin{axis}[bookplot,
    width=7.6cm, height=4.9cm, scale only axis,
    xmin=0, xmax=2.75, ymin=0, ymax=2.75,
    xlabel={volume $V$ ($\SI{e-3}{m^3}$)}, ylabel={pressure $P$ ($\SI{e5}{Pa}$)},
    ylabel style={at={(axis description cs:0,1)},anchor=south west,rotate=-90,xshift=-4pt},
    xtick={1,2}, ytick={1,2}]
  % areas under the two paths
  \fill[cAcc!14] (axis cs:1,0) rectangle (axis cs:2,2);
  \fill[cPrim!18] (axis cs:1,0) rectangle (axis cs:2,1);
  \draw[guide] (axis cs:0,2) -- (axis cs:1,2);
  \draw[guide] (axis cs:0,1) -- (axis cs:1,1);
  \draw[guide] (axis cs:1,0) -- (axis cs:1,1);
  \draw[guide] (axis cs:2,0) -- (axis cs:2,1);
  % path 1 (expand first, at high pressure)
  \draw[curve,cAcc,mid arrow=0.56] (axis cs:1,2) -- (axis cs:2,2);
  \draw[curve,cAcc,mid arrow=0.56] (axis cs:2,2) -- (axis cs:2,1);
  % path 2 (drop the pressure first)
  \draw[curve,cPrim,mid arrow=0.56] (axis cs:1,2) -- (axis cs:1,1);
  \draw[curve,cPrim,mid arrow=0.56] (axis cs:1,1) -- (axis cs:2,1);
  \node[dot] at (axis cs:1,2) {};
  \node[dot] at (axis cs:2,1) {};
  \node[anchor=south east,inner sep=2pt] at (axis cs:1,2) {$i$};
  \node[anchor=north west,inner sep=2pt] at (axis cs:2,1) {$f$};
  \node[cAcc,anchor=south west,inner sep=2pt] at (axis cs:2.03,2.03) {path 1};
  \node[cPrim,anchor=east,inner sep=2pt] at (axis cs:0.97,1.5) {path 2};
  \node[cAcc] at (axis cs:1.5,1.5) {$W_1-W_2$};
  \node[cPrim] at (axis cs:1.5,0.5) {$W_2=\SI{100}{J}$};
  % brace-like dimension: path 1's area is both regions
  \draw[dim,cAcc] (axis cs:2.3,0) -- (axis cs:2.3,2);
  \node[cAcc,anchor=west,align=left] at (axis cs:2.33,0.75) {$W_1$\\$=\SI{200}{J}$};
\end{axis}
\end{tikzpicture}
