% decaylaw_successive: activities in a two-member chain X -> Y -> stable (Vol III Ch 3, fig:decaylaw-successive)
% Pure parent at t = 0.  Time in units of the daughter half-life T_Y; activity in units of A_X(0).
%   A_X = e^{-lX t},   A_Y = lY N_Y = lY/(lY - lX) * (e^{-lX t} - e^{-lY t}) * A_X(0)   (Bateman).
% Panels: T_X/T_Y = 100 (secular), 4 (transient), 1/4 (no equilibrium); lY = ln 2.
% The curves cross where dN_Y/dt = 0, at t_max = ln(lY/lX)/(lY - lX):
%   6.71, 2.67, 0.667 T_Y.  Transient ratio A_Y/A_X -> T_X/(T_X - T_Y) = 4/3.
% (The source figure drew the third panel in units of T_X while labelling the axis t/T_Y;
%  corrected here.)
\begin{tikzpicture}
\begin{groupplot}[bookplot,
    group style={group size=3 by 1, horizontal sep=9mm},
    width=5.55cm, height=4.6cm, ymin=0, ymax=1.32, samples=120,
    xlabel={$t/T_Y$},
    title style={font=\small,text=cInk,yshift=-2pt}]
  % secular: T_X = 100 T_Y
  \nextgroupplot[xmin=0, xmax=8.6, domain=0:8, xtick={2,4,6,8},
      ytick={0.5,1}, yticklabels={$\frac12$,$1$},
      ylabel={$\Act/\Act_X(0)$},
      title={secular: $T_X=100\,T_Y$}]
  \addplot[cPrim] {exp(-ln(2)*x/100)};
  \addplot[cAcc] {(ln(2)/(ln(2)-ln(2)/100))*(exp(-ln(2)*x/100)-exp(-ln(2)*x))};
  \node[dot] at (axis cs:6.712,{exp(-ln(2)*6.712/100)}) {};
  \node[cPrim,anchor=south] at (axis cs:2.4,1.0) {parent X};
  \node[cAcc,anchor=north west] at (axis cs:1.4,0.55) {daughter Y};
  % transient: T_X = 4 T_Y
  \nextgroupplot[xmin=0, xmax=10.8, domain=0:10, xtick={2,4,6,8,10},
      ytick={0.5,1}, yticklabels={$\frac12$,$1$},
      title={transient: $T_X=4\,T_Y$}]
  \addplot[cPrim] {exp(-ln(2)*x/4)};
  \addplot[cAcc] {(4/3)*(exp(-ln(2)*x/4)-exp(-ln(2)*x))};
  \node[dot] at (axis cs:2.667,{exp(-ln(2)*2.667/4)}) {};
  \node[cAcc,anchor=south west,align=left] at (axis cs:4.3,0.55) {$\Act_Y\to\frac43\Act_X$};
  % no equilibrium: T_X = T_Y/4
  \nextgroupplot[xmin=0, xmax=5.4, domain=0:5, xtick={1,2,3,4,5},
      ytick={0.5,1}, yticklabels={$\frac12$,$1$},
      title={none: $T_X=\frac14T_Y$}]
  \addplot[cPrim] {exp(-4*ln(2)*x)};
  \addplot[cAcc] {(1/3)*(exp(-ln(2)*x)-exp(-4*ln(2)*x))};
  \node[dot] at (axis cs:0.6667,{exp(-4*ln(2)*0.6667)}) {};
  \node[cAcc,anchor=south west,align=left] at (axis cs:1.6,0.30) {daughter decays\\ with its own $T_Y$};
\end{groupplot}
\end{tikzpicture}
