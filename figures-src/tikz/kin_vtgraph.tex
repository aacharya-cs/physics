% kin_vtgraph: slope and area of a velocity-time graph (Ch 2, fig:vtgraph)
% v(t) = 1 + 3(1 - exp(-t/3.5)) on 0 <= t <= 8: velocity rising ever more slowly.
% Tangent at t0 = 3: v0 = 2.727, a = dv/dt = (3/3.5) exp(-3/3.5) = 0.364.
% The slope triangle (Delta t = 1.6) sits above the tangent, clear of the curve,
% which (being concave) lies below its tangent.  The shaded area under the curve,
% down to v = 0, is the displacement s over 0 <= t <= 8.
\begin{tikzpicture}
\begin{axis}[bookplot,
    width=8.2cm, height=4.8cm, scale only axis,
    xmin=0, xmax=9, ymin=0, ymax=4.6,
    xlabel={time $t$}, ylabel={velocity $v$},
    xlabel style={at={(axis description cs:1,0)},anchor=north east},
    ylabel style={at={(axis description cs:0,1)},anchor=south west,rotate=-90,xshift=-4pt}]
  \pgfmathdeclarefunction{vf}{1}{\pgfmathparse{1+3*(1-exp(-#1/3.5))}}
  \def\tz{3}
  \pgfmathsetmacro\vz{vf(\tz)}
  \pgfmathsetmacro\az{(3/3.5)*exp(-\tz/3.5)}
  % area under the curve
  \addplot[draw=none,fill=cSlope,domain=0:8] {vf(x)} \closedcycle;
  \draw[guide] (axis cs:8,0) -- (axis cs:8,{vf(8)});
  \node at (axis cs:4.6,1.0) {area $=$ displacement $s$};
  % the velocity curve
  \addplot[cFor,domain=0:8] {vf(x)};
  % tangent and slope triangle
  \def\dt{1.6}
  \addplot[construction,cPrim,domain={\tz-2.3}:{\tz+2.6},samples=2] {\vz+\az*(x-\tz)};
  \draw[cPrim,hair line] (axis cs:{\tz-\dt},{\vz-\az*\dt}) -- (axis cs:{\tz-\dt},\vz) -- (axis cs:\tz,\vz);
  \node[cPrim,anchor=east,inner sep=2pt] at (axis cs:{\tz-\dt},{\vz-\az*\dt/2}) {$\Delta v$};
  \node[cPrim,anchor=south,inner sep=2pt] at (axis cs:{\tz-\dt/2},\vz) {$\Delta t$};
  \node[dot,cPrim] at (axis cs:\tz,\vz) {};
  \node[cPrim,anchor=north west,inner sep=2pt] at (axis cs:{\tz+0.25},{\vz-0.05})
    {slope $=\dfrac{\Delta v}{\Delta t}=a$};
\end{axis}
\end{tikzpicture}
