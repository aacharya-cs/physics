% kinetic_cylinder: the collision cylinder behind the mean free path (Vol II Ch 5, fig:ktcylinder)
% Molecules are spheres of diameter d.  Two touch when their centres are d apart, so the
% moving molecule hits every molecule whose CENTRE lies within distance d of its path:
% a cylinder of radius d (cross-section pi d^2, not pi d^2/4).  In time t it sweeps a
% length (mean relative speed) x t, so it meets  n_V * pi d^2 * (that length)  molecules.
% Centres inside the cylinder (filled) are hit; centres outside (ghosted) are missed.
\begin{tikzpicture}
  \def\r{0.9}\def\L{9.0}
  % the cylinder: two end ellipses in perspective and the two generators
  \draw[guide] (0,\r) -- (\L,\r);
  \draw[guide] (0,-\r) -- (\L,-\r);
  \draw[guide] (0,0) ellipse[x radius={0.3*\r}, y radius=\r];
  \draw[guide] (\L,-\r) arc[start angle=-90,end angle=90,x radius={0.3*\r},y radius=\r];
  \draw[guide,hidden] (\L,\r) arc[start angle=90,end angle=270,x radius={0.3*\r},y radius=\r];
  % the path of the moving molecule's centre
  \draw[thin line,cInk,dash pattern=on 1pt off 1.5pt] (0,0) -- (\L,0);
  % the moving molecule (diameter d drawn to scale: radius r/2)
  \shade[ball color=cBlock] (0,0) circle[radius={\r/2}];
  \draw[cSteel,thin line] (0,0) circle[radius={\r/2}];
  \draw[velocity] (0.55,0) -- (1.6,0) node[above,inner sep=2pt] {$\bar v$};
  % radius d of the cylinder
  \draw[dim] (-0.75,0) -- (-0.75,\r) node[midway,left,cSteel] {$d$};
  \draw[dimtick] (-0.85,\r) -- (0,\r);
  \draw[dimtick] (-0.85,0) -- (-0.45,0);
  % targets whose centres lie inside: hit (with the contact at distance d)
  \foreach \x/\y in {3.2/0.55, 5.6/-0.7, 7.4/0.25}{
    \shade[ball color=cAcc!60] (\x,\y) circle[radius={\r/2}];
    \draw[cAcc,thin line] (\x,\y) circle[radius={\r/2}];
    \node[dot,cInk,minimum size=2pt] at (\x,\y) {};
  }
  % targets whose centres lie outside: missed
  \foreach \x/\y in {2.1/-1.45, 4.5/1.5, 6.6/1.3, 8.3/-1.5}{
    \begin{scope}[ghost]
      \shade[ball color=cMoon] (\x,\y) circle[radius={\r/2}];
      \draw[cSteel,thin line] (\x,\y) circle[radius={\r/2}];
    \end{scope}
    \node[dot,cGrey,minimum size=2pt] at (\x,\y) {};
  }
  % length swept
  \draw[dim] (0,-2.15) -- (\L,-2.15) node[midway,fill=white,text=cSteel] {length swept relative to the others in time $t$: $\sqrt2\,\bar v\,t$};
  % labels
  \node[cAcc,anchor=west,font=\footnotesize] at (\L+0.45,0.25) {centre inside: hit};
  \node[cSteel,anchor=west,font=\footnotesize] at (\L+0.45,-0.25) {centre outside: missed};
  \node[cSteel,anchor=south west,font=\footnotesize] at (0.2,1.0) {cross-section $\pi d^2$};
\end{tikzpicture}
