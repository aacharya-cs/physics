% kinetic_wall: one molecule in a cube, and one elastic hit on a wall (Vol II Ch 5, fig:ktwall)
% Left: a cube of side ell; the shaded wall at x = ell has area ell^2.  A molecule with
%   velocity v (components v_x, v_y) crosses to the far wall and back, a distance 2 ell
%   at speed |v_x|, so it returns every 2 ell/|v_x|.
% Right: the hit, seen from above.  The wall is smooth and very massive, so the
%   component normal to it reverses (v_x -> -v_x) and v_y is untouched (Vol I, Ch 12:
%   an elastic bounce off a fixed wall).  Momentum change of the molecule: -2 m v_x;
%   the wall receives +2 m v_x per hit, shown as the force on the wall.
\begin{tikzpicture}
  % ---------------------------------------------------------------- the cube
  \def\s{3.3}\def\dx{1.25}\def\dy{0.8}
  % shaded wall x = ell (right face)
  \fill[cPrim!12] (\s,0) -- (\s+\dx,\dy) -- (\s+\dx,\s+\dy) -- (\s,\s) -- cycle;
  % hidden edges
  \draw[outline,hidden,thin line] (\dx,\dy) -- (\s+\dx,\dy);
  \draw[outline,hidden,thin line] (\dx,\dy) -- (\dx,\s+\dy);
  \draw[outline,hidden,thin line] (\dx,\dy) -- (0,0);
  % visible edges
  \draw[outline] (0,0) rectangle (\s,\s);
  \draw[outline] (0,\s) -- (\dx,\s+\dy) -- (\s+\dx,\s+\dy) -- (\s,\s);
  \draw[outline] (\s,0) -- (\s+\dx,\dy) -- (\s+\dx,\s+\dy);
  \node[cPrim,align=center] at (\s+0.66,1.8) {area\\ $\ell^2$};
  % side length
  \draw[dim] (0,-0.3) -- (\s,-0.3) node[midway,below,cSteel] {$\ell$};
  % axes
  \draw[-smallhead,thin line] (-1.2,0.3) -- (-0.65,0.3) node[right,inner sep=1pt] {$x$};
  \draw[-smallhead,thin line] (-1.2,0.3) -- (-1.2,0.85) node[above,inner sep=1pt] {$y$};
  % a few other molecules
  \foreach \x/\y/\a in {2.7/0.45/35, 0.75/2.6/-30, 2.75/2.75/60, 0.5/0.6/200, 2.9/1.9/290}{
    \pic at (\x,\y) {particle};
    \draw[velocity,thin line,-smallhead] (\x,\y) ++(\a:0.16) -- ++(\a:0.42);
  }
  % the molecule followed, with its velocity and x-component
  \pic at (1.0,1.65) {particle};
  \draw[velocity] (1.13,1.72) -- (2.25,2.25) node[above,inner sep=2pt] {$\vv$};
  \draw[component,cFor] (1.13,1.6) -- (2.25,1.6) node[below,inner sep=2pt] {$v_x$};
  % the round trip
  \draw[guide,-smallhead] (0.2,1.25) -- (\s-0.15,1.25);
  \draw[guide,-smallhead] (\s-0.15,1.2) -- (0.2,1.2);
  \node[cSteel,font=\footnotesize] at (1.7,0.95) {round trip $2\ell$};
  % ------------------------------------------------------------ the one hit
  \begin{scope}[xshift=7.6cm,yshift=0.2cm]
    \fill[cRock] (1.6,0) rectangle (1.85,3.0);
    \draw[outline] (1.6,0) -- (1.6,3.0);
    \node[anchor=west] at (1.95,2.85) {wall at $x=\ell$};
    % incoming and outgoing
    \pic at (1.47,1.5) {particle};
    \draw[velocity] (-0.4,0.3) -- (1.3,1.4) node[pos=0.45,above left,inner sep=1pt] {before};
    \draw[velocity] (1.3,1.6) -- (-0.4,2.7) node[pos=0.55,above right,inner sep=1pt] {after};
    \draw[component,cFor] (-0.4,0.12) -- (1.3,0.12) node[pos=0.5,below,inner sep=2pt] {$+v_x$};
    \draw[component,cFor] (1.3,2.95) -- (-0.4,2.95) node[pos=0.5,above,inner sep=2pt] {$-v_x$};
    % the force on the wall
    \draw[force] (1.85,1.5) -- (2.75,1.5);
    \node[cCrim,anchor=west,align=left] at (1.95,0.85) {wall gains\\ $2mv_x$ per hit};
    \node[align=left,anchor=west] at (1.95,2.25) {$v_y$ unchanged};
  \end{scope}
\end{tikzpicture}
