% nuc_semfterms: building E_b/A term by term from the liquid-drop formula
% (Vol III Ch 1, fig:nuc-semfterms).  Per nucleon, with Z = Z0(A) continuous,
%   volume      +aV                        aV = 15.75 MeV
%   surface     -aS A^(-1/3)               aS = 17.8
%   Coulomb     -aC Z(Z-1)/A^(4/3)         aC = 0.711
%   asymmetry   -aA (A-2Z)^2/A^2           aA = 23.7
%   Z0(A) = A/(1.98 + 0.0155 A^(2/3)).
% At A = 56: 15.75 - 4.65 - 2.05 - 0.21 = 8.84 MeV (measured 8.79).
% Dots: measured E_b/A for the A >= 9 nuclides of fig:nuc-bea.
\begin{tikzpicture}
\begin{axis}[bookplot,
    width=11.5cm, height=7cm,
    xmin=0, xmax=258, ymin=5, ymax=17.2,
    xlabel={mass number $A$}, ylabel={MeV per nucleon},
    xtick={0,50,100,150,200,250}, ytick={6,8,10,12,14,16},
    declare function={zo(\a)=\a/(1.98+0.0155*\a^(2/3));
      vs(\a)=15.75-17.8*\a^(-1/3);
      vsc(\a)=vs(\a)-0.711*zo(\a)*(zo(\a)-1)/\a^(4/3);
      tot(\a)=vsc(\a)-23.7*(\a-2*zo(\a))^2/\a^2;}]
  \addplot[name path=vol,cTeal,domain=10:250,samples=2] {15.75};
  \addplot[name path=vs,cAmber,domain=10:250] {vs(x)};
  \addplot[name path=vsc,cPlum,domain=10:250] {vsc(x)};
  \addplot[name path=tot,cPrim,domain=10:250] {tot(x)};
  \addplot[cAmber!25] fill between[of=vol and vs];
  \addplot[cPlum!18] fill between[of=vs and vsc];
  \addplot[cGrey!35] fill between[of=vsc and tot];
  \addplot[only marks,mark=*,mark size=1.1pt,cInk] coordinates {
    (9,6.463) (11,6.928) (12,7.680) (14,7.476) (16,7.976) (19,7.779) (20,8.032) (24,8.261)
    (27,8.332) (28,8.448) (32,8.493) (40,8.551) (56,8.790) (62,8.795) (63,8.752) (64,8.736)
    (88,8.733) (90,8.710) (120,8.504) (132,8.428) (138,8.393) (184,8.005) (197,7.916)
    (206,7.875) (208,7.867) (209,7.848) (232,7.615) (235,7.591) (238,7.570)};
  % labels, each in the colour of its curve
  \node[cTeal,anchor=south] at (axis cs:150,15.8) {volume term $+a_V$ (saturation)};
  \node[cAmber!80!black,anchor=west] at (axis cs:110,14.0) {$-$\,surface (fewer neighbours)};
  \node[cPlum,anchor=west] at (axis cs:150,10.3) {$-$\,Coulomb (proton repulsion)};
  \node[cPrim,anchor=north] at (axis cs:150,7.55) {$-$\,asymmetry $=$ total $\Eb/A$};
  \draw[leader] (axis cs:30,6.45) -- (axis cs:36,5.7) node[anchor=west,inner sep=1pt] {measured};
\end{axis}
\end{tikzpicture}
